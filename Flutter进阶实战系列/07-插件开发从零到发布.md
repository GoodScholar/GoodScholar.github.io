## 📊 Package vs Plugin

| 类型 | 包含原生代码？ | 创建命令 | 示例 |
|------|-------------|---------|------|
| **Package** | ❌ 纯 Dart | `flutter create --template=package` | `freezed`, `go_router` |
| **Plugin** | ✅ Dart + iOS/Android | `flutter create --template=plugin` | `camera`, `url_launcher` |
| **FFI Plugin** | ✅ Dart + C/C++ | `flutter create --template=plugin_ffi` | `sqlite3_flutter_libs` |

## 🔌 2. Federated Plugin 架构

大型插件推荐使用 Federated（联邦化）架构：

```
device_battery_plus/                    # App-facing package
├── 定义 Dart API
├── 依赖 platform_interface
│
device_battery_plus_platform_interface/ # 平台接口
├── 定义抽象方法
├── 定义 MethodChannel 默认实现
│
device_battery_plus_android/            # Android 实现
├── 实现 platform_interface
│
device_battery_plus_ios/                # iOS 实现
├── 实现 platform_interface
```

### 平台接口

```dart
// device_battery_plus_platform_interface/lib/
abstract class DeviceBatteryPlusPlatform extends PlatformInterface {
  static DeviceBatteryPlusPlatform _instance = MethodChannelDeviceBatteryPlus();

  static DeviceBatteryPlusPlatform get instance => _instance;
  static set instance(DeviceBatteryPlusPlatform instance) {
    PlatformInterface.verify(instance, _token);
    _instance = instance;
  }

  static final Object _token = Object();

  DeviceBatteryPlusPlatform() : super(token: _token);

  Future<int> getBatteryLevel();
  Future<String> getBatteryState();
  Stream<int> get onBatteryLevelChanged;
}
```

## 📱 3. 实现原生代码

### Dart 端 API

```dart
// lib/device_battery_plus.dart
class DeviceBatteryPlus {
  static const MethodChannel _channel = MethodChannel('device_battery_plus');
  static const EventChannel _eventChannel =
      EventChannel('device_battery_plus/events');

  /// 获取当前电量百分比
  static Future<int> get batteryLevel async {
    final int level = await _channel.invokeMethod('getBatteryLevel');
    return level;
  }

  /// 获取电池状态：charging / discharging / full / unknown
  static Future<BatteryState> get batteryState async {
    final String state = await _channel.invokeMethod('getBatteryState');
    return BatteryState.values.firstWhere(
      (e) => e.name == state,
      orElse: () => BatteryState.unknown,
    );
  }

  /// 监听电量变化
  static Stream<int> get onBatteryLevelChanged {
    return _eventChannel.receiveBroadcastStream().map((e) => e as int);
  }
}

enum BatteryState { charging, discharging, full, unknown }
```

### Android 实现 (Kotlin)

```kotlin
class DeviceBatteryPlusPlugin : FlutterPlugin, MethodCallHandler,
    EventChannel.StreamHandler {
    private lateinit var channel: MethodChannel
    private lateinit var eventChannel: EventChannel
    private lateinit var context: Context

    override fun onAttachedToEngine(binding: FlutterPluginBinding) {
        context = binding.applicationContext
        channel = MethodChannel(binding.binaryMessenger, "device_battery_plus")
        channel.setMethodCallHandler(this)

        eventChannel = EventChannel(binding.binaryMessenger,
            "device_battery_plus/events")
        eventChannel.setStreamHandler(this)
    }

    override fun onMethodCall(call: MethodCall, result: Result) {
        when (call.method) {
            "getBatteryLevel" -> {
                val manager = context.getSystemService(Context.BATTERY_SERVICE)
                    as BatteryManager
                val level = manager.getIntProperty(
                    BatteryManager.BATTERY_PROPERTY_CAPACITY)
                result.success(level)
            }
            "getBatteryState" -> {
                val intent = context.registerReceiver(null,
                    IntentFilter(Intent.ACTION_BATTERY_CHANGED))
                val status = intent?.getIntExtra(
                    BatteryManager.EXTRA_STATUS, -1) ?: -1
                result.success(when (status) {
                    BatteryManager.BATTERY_STATUS_CHARGING -> "charging"
                    BatteryManager.BATTERY_STATUS_DISCHARGING -> "discharging"
                    BatteryManager.BATTERY_STATUS_FULL -> "full"
                    else -> "unknown"
                })
            }
            else -> result.notImplemented()
        }
    }

    override fun onListen(arguments: Any?, events: EventChannel.EventSink) {
        // 注册电量变化广播
    }

    override fun onCancel(arguments: Any?) {}

    override fun onDetachedFromEngine(binding: FlutterPluginBinding) {
        channel.setMethodCallHandler(null)
    }
}
```

### iOS 实现 (Swift)

```swift
public class DeviceBatteryPlusPlugin: NSObject, FlutterPlugin {
    public static func register(with registrar: FlutterPluginRegistrar) {
        let channel = FlutterMethodChannel(
            name: "device_battery_plus",
            binaryMessenger: registrar.messenger())
        let instance = DeviceBatteryPlusPlugin()
        registrar.addMethodCallDelegate(instance, channel: channel)
    }

    public func handle(_ call: FlutterMethodCall,
                        result: @escaping FlutterResult) {
        switch call.method {
        case "getBatteryLevel":
            UIDevice.current.isBatteryMonitoringEnabled = true
            let level = Int(UIDevice.current.batteryLevel * 100)
            result(level)
        case "getBatteryState":
            UIDevice.current.isBatteryMonitoringEnabled = true
            let state: String = switch UIDevice.current.batteryState {
            case .charging: "charging"
            case .full: "full"
            case .unplugged: "discharging"
            default: "unknown"
            }
            result(state)
        default:
            result(FlutterMethodNotImplemented)
        }
    }
}
```

## 📝 4. 文档与 Example

### README 必备内容

```markdown
# device_battery_plus

获取设备电池信息的 Flutter 插件。

## 功能
- ✅ 获取电量百分比
- ✅ 获取充电状态
- ✅ 监听电量变化

## 安装
``yaml
dependencies:
  device_battery_plus: ^1.0.0
``

## 使用
``dart
// 获取电量
final level = await DeviceBatteryPlus.batteryLevel;

// 监听变化
DeviceBatteryPlus.onBatteryLevelChanged.listen((level) {
  print('电量: $level%');
});
``

## 平台支持
| Android | iOS |
|---------|-----|
| ✅      | ✅  |
```
